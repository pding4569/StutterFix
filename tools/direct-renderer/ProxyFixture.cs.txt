  image.ReadPixels(new Rect(0,0,rt.width,rt.height),0,0); image.Apply();
  var data=image.GetPixels32(); UnityEngine.Object.DestroyImmediate(image); RenderTexture.active=old; return data;

 static int MaxDelta(Color32[] a,Color32[] b) {int n=0;for(int k=0;k<a.Length;k++)for(int c=0;c<4;c++)n=Math.Max(n,Math.Abs(a[k][c]-b[k][c]));return n;}
 static float Blend(float b,float s,int mode) {
  switch(mode) {
   case 1:return Math.Min(b,s);case 2:return b*s;case 4:return b+s-1;case 6:return Math.Max(b,s);
   case 7:return 1-(1-b)*(1-s);case 9:return b+s;case 11:return b>.5f?1-2*(1-b)*(1-s):2*b*s;
   case 12:return b*(1-b)*s+(1-(1-b)*(1-s))*b;case 18:return Math.Abs(b-s);case 19:return b+s-2*b*s;
   case 20:return b-s;case 21:return Mathf.Clamp01(b/s);default:return s;
  }
 }
